# Target the policy ABI supported by HAOS AppArmor 3.x and newer parsers.
# ABI 3.0 permits user namespaces implicitly; the userns rule requires 4.x.
abi <abi/3.0>,

#include <tunables/global>

# No blanket file/execute rule: root setup is confined separately from Chromium.
profile app_kiosk flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix,
  network netlink raw,
  capability chown,
  capability dac_override,
  capability fowner,
  capability fsetid,
  capability setuid,
  capability setgid,
  capability kill,
  signal (send,receive),
  ptrace (read),

  /init rix,
  /bin/{bash,busybox,sh} rix,
  /usr/bin/{bash,env,id,jq,date,python3,python3.*,curl,head,grep,sed,cut,ls,cat,chmod,chown,mkdir,seq,sleep,readlink} rix,
  /usr/bin/{seatd,wlr-randr,wtype,swayidle,vainfo,certutil,openssl,killall} rix,
  /usr/local/bin/cage rix,
  /usr/lib/bashio/** rix,
  /usr/bin/bashio rix,
  /run.sh rix,
  /app/ r,
  /app/** r,
  /app/launch-browser.sh rix,
  /usr/lib/** mr,
  /lib/** mr,
  /usr/share/** r,
  /etc/** r,
  /etc/chromium{,-browser}/policies/managed/ rw,
  /etc/chromium{,-browser}/policies/managed/*.json rw,
  /etc/opt/chrome/policies/managed/ rw,
  /etc/opt/chrome/policies/managed/*.json rw,
  /data/ rw,
  /data/options.json rw,
  /data/browser-home{,/**} rwk,
  /data/chromium-profile{,/**} rwk,
  /tmp/{,**} rwkm,
  /run/seatd.sock rw,
  /run/udev/** r,
  /run/audio/** rw,
  /proc/{,**} r,
  /sys/{,**} r,
  /dev/dri/ r,
  /dev/dri/* rw,
  /dev/input/{,by-id/,by-path/} r,
  /dev/input/{event*,by-id/*,by-path/*} rw,
  /dev/shm/{,**} rwkm,
  /dev/tty rw,

  # Base-image init and s6 services execute before the app's root setup.
  /package/** rix,
  /command/** rix,
  /run/{s6,s6-rc*,service}/{,**} rwix,
  /etc/{cont-init.d,cont-finish.d,services.d}/{,**} rix,
  /run/{,**} rwk,

  /usr/lib/chromium/chromium cx -> browser,
  profile browser flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network unix,
    signal (send,receive),
    ptrace (read,trace),
    # Chromium creates its own user/PID/network namespaces and seccomp sandbox.
    # These profile permissions do not grant host capabilities to its uid.
    capability sys_admin,
    capability sys_chroot,
    capability setuid,
    capability setgid,
    mount fstype=proc -> /proc/,
    umount /proc/,
    /usr/lib/chromium/** rixm,
    /usr/lib/** mr,
    /lib/** mr,
    /usr/share/** r,
    /etc/{hosts,hostname,resolv.conf,nsswitch.conf,passwd,group,localtime,timezone} r,
    /etc/{fonts,ssl,chromium,chromium-browser,opt/chrome}/{,**} r,
    /data/browser-home{,/**} rwk,
    /data/chromium-profile{,/**} rwk,
    /tmp/{,**} rwkm,
    /dev/shm/{,**} rwkm,
    /dev/dri/ r,
    /dev/dri/renderD* rw,
    /proc/{,**} r,
    owner /proc/*/{oom_score_adj,uid_map,gid_map,setgroups} w,
    owner /proc/*/task/*/comm rw,
    /sys/{,**} r,
    /run/audio/** rw,
    /run/udev/** r,
    # No access to /data/options.json, root credentials, or shared HA storage.
  }
}
